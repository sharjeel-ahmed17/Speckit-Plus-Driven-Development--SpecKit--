# docker file for next js : 
```
FROM node:20-alpine
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npm run build
EXPOSE 7860
ENV HOSTNAME=0.0.0.0
ENV PORT=7860
ENV NODE_ENV=production
CMD ["npm", "start"]
```

# docker file for python 

```
FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .
EXPOSE 7860
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "7860"]
```
## docker file for python uv
```
# install python
FROM python:3.12-slim

# install uv and copy specific file to run uv
COPY --from=ghcr.io/astral-sh/uv:latest /uv /uvx /bin/

# where all my project files are store
WORKDIR /app

# now copy all my project files into /app
COPY . /app

# Expose PORT
# EXPOSE 8000

RUN uv sync --frozen --no-cache

# Run the application.
CMD ["uv", "run","uvicorn", "main:app", "--host", "0.0.0.0","--port", "7896"]
```


## kubectl envirment varaibles : 

```
# backend secret
kubectl create secret generic todo-backend-secret \
  --from-literal=BETTER_AUTH_SECRET='your-key' \
  --from-literal=DATABASE_URL='database-url' \
  --from-literal=COHERE_API_KEY='api-key'


#   forntend secret


kubectl create secret generic todo-frontend-secret \
  --from-literal=BETTER_AUTH_SECRET='your-key' \
  --from-literal=NEXT_PUBLIC_BETTER_AUTH_URL='http://localhost:8000' \
  --from-literal=NEXT_PUBLIC_API_URL='http://localhost:8000/api'

```

## create image without your docker hub username
